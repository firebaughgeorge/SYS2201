# create a new column age_group based on the criteria above
bins = [0, 12, 20, 40, 60, 150]
labels = ['child', 'teen', 'adult', 'middle-aged', 'senior']
titanic_data['age_group'] = pd.cut(titanic_data['age'], bins = bins, labels = labels)

# Calculate survival rates by age group
survival_by_age = titanic_data.groupby('age_group', observed=True)['survived'].mean()
print(survival_by_age)
